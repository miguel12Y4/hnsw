cmake_minimum_required(VERSION 3.14)
project(HNSW CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include_directories(include)
add_library(hnsw include/hnsw_index.cc)
target_include_directories(hnsw PUBLIC include)
add_executable(example examples/main.cpp)
target_link_libraries(example hnsw)
add_executable(bench examples/bench.cpp)
target_link_libraries(bench hnsw)

enable_testing()
add_executable(hnsw_tests tests/test_hnsw.cpp)
target_link_libraries(hnsw_tests hnsw)
add_test(NAME hnsw_tests COMMAND hnsw_tests)
